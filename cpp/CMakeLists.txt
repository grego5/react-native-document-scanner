cmake_minimum_required(VERSION 3.22.1)
project(DocumentScannerCore LANGUAGES CXX)
option(DOCUMENT_SCANNER_BUILD_TESTS "Build standalone portable core tests" OFF)

add_library(document_scanner_core STATIC src/Pixels.cpp src/Tensor.cpp src/Geometry.cpp src/RenderGeometry.cpp src/PageEdges.cpp src/DocumentFilters.cpp src/GrayscalePng.cpp src/PdfWriter.cpp src/Motion.cpp src/FlowEstimator.cpp src/FieldKernels.cpp src/PageEdgesRemap.cpp)
target_compile_features(document_scanner_core PUBLIC cxx_std_20)
set_target_properties(document_scanner_core PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
target_include_directories(document_scanner_core PUBLIC include PRIVATE src)
# Validated normalization and field arithmetic must keep separate rounding.
if(MSVC)
  target_compile_options(document_scanner_core PRIVATE /fp:strict)
else()
  set_source_files_properties(src/Geometry.cpp src/RenderGeometry.cpp src/PageEdges.cpp src/DocumentFilters.cpp src/PdfWriter.cpp src/Motion.cpp src/FlowEstimator.cpp PROPERTIES COMPILE_OPTIONS "-ffp-contract=off")
  set_source_files_properties(src/Tensor.cpp PROPERTIES COMPILE_OPTIONS "-O2;-ffp-contract=off")
  # Keep existing Android debug kernel optimization and float rounding.
  set_source_files_properties(src/FieldKernels.cpp PROPERTIES COMPILE_OPTIONS "-O2;-ffp-contract=off")
  set_source_files_properties(src/PageEdgesRemap.cpp PROPERTIES COMPILE_OPTIONS "-O2")
endif()

find_package(ZLIB REQUIRED)
target_link_libraries(document_scanner_core PRIVATE ZLIB::ZLIB)

if(ANDROID)
  if(NOT TARGET OpenCV::opencv_java4)
    message(FATAL_ERROR "Android core requires the existing OpenCV::opencv_java4 runtime")
  endif()
  target_link_libraries(document_scanner_core PRIVATE OpenCV::opencv_java4)
else()
  find_package(OpenCV 4.12 REQUIRED CONFIG COMPONENTS core imgproc video)
  target_link_libraries(document_scanner_core PRIVATE ${OpenCV_LIBS})
  target_include_directories(document_scanner_core PRIVATE ${OpenCV_INCLUDE_DIRS})
endif()
